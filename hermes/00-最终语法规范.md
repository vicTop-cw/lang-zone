## 一、变量与绑定

```lz
x = 1                    # 类型推断，默认可变（copy-by-default）
x: int = 1               # 显式类型
mut x = 1                # 可变
mut x: int = 1           # 可变+类型
const PI = 3.14          # 编译期常量

# 引用
ref r = &x               # 不可变借用
mut ref r = &mut x       # 可变借用

# 堆分配（统一一种语法）
ptr = alloc(int, 100)    # 分配 100 个 int
ptr[0] = 42              # 下标
free(ptr)                # 手动释放
```

**删除：** `global`, `&raw`, `OwnedPtr`, `vec!`

> 注：`let` 在编译器 `src/*.rs` 中**保留**（用于 `guard let` 与顶层绑定），不删除。

## 二、函数

```lz
# 基本
def add(a: int, b: int)-> int =
    return a + b

def add(a: int, b: int)-> int =
    a + b               # 表达式体，自动 return

def greet(name: str) =
    print(f"Hello, {name}")

# 参数默认值
def power(base: int, exp: int = 2)-> int =
    if exp == 0: 1
    else: base * power(base, exp - 1)

# 可变参数（末尾 List<T> 自动压栈）
def sum(nums: int, rest: List<int>)-> int =
    total = nums
    for n in rest: total += n
    return total

sum(1, 2, 3, 4)          # nums=1, rest=[2,3,4] → total=10
```

**删除：** `->` 返回值语法，`fn(T)->U` 函数类型

## 三、类型

```lz
int          # 整数（等于 Rust i64）
f64          # 浮点
str          # 字符串
bool         # 布尔
None         # 空

List<T>      # 列表: [1, 2, 3]
Dict<K, V>   # 字典: {"a": 1}
Set<T>       # 集合: {1, 2, 3}
Tuple[A, B]  # 元组: (42, "hi")
Option<T>    # 可选: Some(x) | None
Result<T,E>  # 结果: Ok(x) | Err(e)

# Option 操作
user?.name               # 安全导航
user?.name ?? "guest"    # null 合并

# Option 语法糖：T? ≡ Option<T>
x: int? = None           # 等于 x: Option<int> = None
name: str? = Some("hi")  # 等于 name: Option<str> = Some("hi")
val?                     # 类型标注中的 ? 统一表示 Option<T>
```

**删除：** `[int]` `{:str: int}` `{int}`（容器类型统一用 `List<T>` `Dict<K,V>` `Set<T>`）

## 四、泛型

```lz
# 泛型函数
def identity<T>(x: T)-> T = x

# 泛型结构体
struct Box<T> = value: T

# 约束
def max<T>(a: T, b: T)-> T where T <: Ord =
    if a > b: a
    else: b

def clone_and_sort<T>(items: List<T>)-> List<T> 
    where T <: Clone + Ord =
    items |> map(|x| x.clone()) |> sorted

# 约束符号
T <: Clone            # 单约束
T <: Clone + Ord      # 多约束
T <: Any              # 无约束
T <: Copy             # 值语义
T <: Default          # 无参构造
T <: Hash             # 可哈希

# 函数类型
def apply<T, U>(f: (T) -> U, x: T)-> U where T <: Any = f(x)
```

**删除：** `where T : Ord`（冒号改 `<:`），`fn(T) -> U`（括号写法）

## 五、struct / enum / trait / impl

```lz
struct Point =
    x: f64
    y: f64

    def distance(self: Point, other: Point)-> f64 =
        dx = self.x - other.x
        dy = self.y - other.y
        (dx * dx + dy * dy).sqrt()

enum Option<T> =
    None
    Some(T)

enum Result<T, E> =
    Ok(T)
    Err(E)

trait Drawable =
    def draw(self: Self) = ...          # 抽象方法：必须有"= ..."（省略号表示待实现）

trait Summary =
    def summarize(self: Self)-> str = ...  # 抽象方法必须有 '=' 
    def default_method(self: Self) =      # 默认实现：有 "=" 后跟缩进体
        print("default impl")

impl Drawable for Circle =
    def draw(self: Circle) =              # 实现 trait 抽象方法
        print(f"Drawing circle")
```

**规则：**
- **抽象方法必须写在 trait 中**，且必须以 `= ...` 结尾（`...` 表示待实现）
- **所有类型参数统一使用尖括号** `<>`，不使用方括号 `[]`：`List<T>` `Dict<K,V>` `Set<T>` `Option<T>` `Result<T,E>`

**删除：** `extend` 关键字（用 `impl`），`->` 返回值

## 六、控制流

```lz
if x > 0:
    "positive"
elif x == 0:
    "zero"
else:
    "negative"

for i in 0..10:
    print(i)

while x > 0:
    x -= 1

loop:
    if x > 5: break

match r:
    case Ok(v) => print(v)
    case Err(e) => print(e)
    case _ => print("?")

# guard 表达式（提前终止，else 后直接跟表达式，不写 return）
def safe_divide(a: int, b: int)-> Option<int> =
    guard b != 0 else: None        # ← 不写 return，直接写值
    Some(a / b)

# 上面等价于：
# if b == 0 { return None; }
# Some(a / b)

# with 块
with Resource() as r:
    r.use()
```

**删除：** `raises`，`out`
